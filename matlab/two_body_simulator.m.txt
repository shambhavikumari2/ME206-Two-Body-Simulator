function two_body_simulator()
    % Interactive Two-Body Gravitational Simulator (Center-of-Mass Frame)
    % Integrator: Velocity Verlet (symplectic, preserves orbital energy)

    %% 1. Create Main Figure & Grid Layout
    fig = uifigure('Name', 'Interactive Two-Body Gravitational Simulator', ...
                   'Position', [100, 100, 1050, 620], ...
                   'Color', [0.12, 0.12, 0.14]);

    mainGrid = uigridlayout(fig, [1, 2]);
    mainGrid.ColumnWidth = {'2.5x', '1x'};
    mainGrid.Padding = [12, 12, 12, 12];
    mainGrid.ColumnSpacing = 12;

    %% 2. Left Panel: Orbital Simulation Axes
    ax = uiaxes(mainGrid);
    ax.BackgroundColor = [0.06, 0.07, 0.09];
    ax.XColor = [0.75, 0.75, 0.78];
    ax.YColor = [0.75, 0.75, 0.78];
    ax.GridColor = [0.25, 0.25, 0.30];
    ax.GridAlpha = 0.5;
    ax.XGrid = 'on';
    ax.YGrid = 'on';
    axis(ax, 'equal');
    xlim(ax, [-3.5, 3.5]);
    ylim(ax, [-3.5, 3.5]);
    xlabel(ax, 'x (AU-scaled)', 'Color', [0.85, 0.85, 0.85]);
    ylabel(ax, 'y (AU-scaled)', 'Color', [0.85, 0.85, 0.85]);
    title(ax, 'Two-Body Orbital Dynamics (Barycentric Frame)', ...
          'Color', 'w', 'FontSize', 13);
    hold(ax, 'on');

    % Barycenter marker at origin (0,0)
    plot(ax, 0, 0, '+w', 'MarkerSize', 10, 'LineWidth', 1.3, ...
         'DisplayName', 'Barycenter (CoM)');

    % Animated orbital trails
    trail1 = animatedline(ax, 'Color', [1.0, 0.65, 0.15], ...
                          'LineWidth', 1.5, 'MaximumNumPoints', 700, ...
                          'DisplayName', 'Body 1 Trail');
    trail2 = animatedline(ax, 'Color', [0.25, 0.75, 1.0], ...
                          'LineWidth', 1.5, 'MaximumNumPoints', 700, ...
                          'DisplayName', 'Body 2 Trail');

    % Body markers
    body1 = plot(ax, 0, 0, 'o', 'MarkerSize', 14, ...
                 'MarkerFaceColor', [1.0, 0.65, 0.15], ...
                 'MarkerEdgeColor', 'w', 'DisplayName', 'Body 1 (m1)');
    body2 = plot(ax, 0, 0, 'o', 'MarkerSize', 9, ...
                 'MarkerFaceColor', [0.25, 0.75, 1.0], ...
                 'MarkerEdgeColor', 'w', 'DisplayName', 'Body 2 (m2)');

    lgd = legend(ax, 'Location', 'northeast');
    lgd.TextColor = 'w';
    lgd.Color = [0.14, 0.15, 0.18];

    %% 3. Right Panel: Controls & Live Telemetry
    ctrlPanel = uipanel(mainGrid, 'Title', 'Controls & Conservation Telemetry', ...
                        'BackgroundColor', [0.16, 0.17, 0.20], ...
                        'ForegroundColor', 'w', 'FontWeight', 'bold', ...
                        'FontSize', 12);

    ctrlGrid = uigridlayout(ctrlPanel, [11, 1]);
    ctrlGrid.RowHeight = {22, 42, 22, 42, 36, 28, 24, 24, 24, 24, '1x'};
    ctrlGrid.Padding = [12, 12, 12, 12];
    ctrlGrid.RowSpacing = 8;

    % Mass Ratio Slider
    lblMass = uilabel(ctrlGrid, 'Text', 'Mass Ratio (m2 / m1): 0.25', ...
                      'FontColor', 'w', 'FontWeight', 'bold');
    sldMass = uislider(ctrlGrid, 'Limits', [0.05, 1.0], 'Value', 0.25);

    % Initial Velocity Multiplier Slider
    lblVel = uilabel(ctrlGrid, 'Text', 'Initial Speed (v / v_circ): 1.00', ...
                     'FontColor', 'w', 'FontWeight', 'bold');
    sldVel = uislider(ctrlGrid, 'Limits', [0.40, 1.50], 'Value', 1.00);

    % Pause / Reset Button Row
    btnGrid = uigridlayout(ctrlGrid, [1, 2]);
    btnGrid.Padding = [0, 0, 0, 0];
    btnPause = uibutton(btnGrid, 'Text', 'Pause', ...
                        'BackgroundColor', [0.25, 0.45, 0.75], ...
                        'FontColor', 'w', 'FontWeight', 'bold');
    btnReset = uibutton(btnGrid, 'Text', 'Reset Orbit', ...
                        'BackgroundColor', [0.30, 0.32, 0.36], ...
                        'FontColor', 'w', 'FontWeight', 'bold');

    % Live Telemetry Labels
    uilabel(ctrlGrid, 'Text', '--- Live Conserved Quantities ---', ...
            'FontColor', [0.7, 0.85, 1.0], 'FontWeight', 'bold', ...
            'HorizontalAlignment', 'center');
    lblRegime = uilabel(ctrlGrid, 'Text', 'Orbit Type: Circular (E < 0)', ...
                        'FontColor', [0.4, 1.0, 0.5], 'FontWeight', 'bold');
    lblKE = uilabel(ctrlGrid, 'Text', 'Kinetic Energy (K): 0.0000', ...
                    'FontColor', 'w');
    lblPE = uilabel(ctrlGrid, 'Text', 'Potential Energy (U): 0.0000', ...
                    'FontColor', 'w');
    lblTE = uilabel(ctrlGrid, 'Text', 'Total Energy (E): 0.0000', ...
                    'FontColor', [1.0, 0.85, 0.35], 'FontWeight', 'bold');
    lblAngMom = uilabel(ctrlGrid, 'Text', 'Angular Momentum (L): 0.0000', ...
                        'FontColor', 'w', 'VerticalAlignment', 'top');

    %% 4. Simulation State Struct
    state.G = 1.0;
    state.m1 = 10.0;
    state.r0 = 2.0;
    state.dt = 0.001;
    state.substeps = 15;
    state.paused = false;

    resetSimulation();

    %% 5. Assign UI Callbacks
    sldMass.ValueChangedFcn = @(~, ~) onSliderChange();
    sldVel.ValueChangedFcn  = @(~, ~) onSliderChange();
    btnReset.ButtonPushedFcn = @(~, ~) resetSimulation();
    btnPause.ButtonPushedFcn = @(~, ~) togglePause();

    %% 6. Non-Blocking Timer Loop (Smooth UI & Clean Exit)
    simTimer = timer('ExecutionMode', 'fixedRate', ...
                     'Period', 0.02, ...
                     'TimerFcn', @(~, ~) stepSimulation());
    fig.CloseRequestFcn = @(~, ~) cleanupAndClose();
    start(simTimer);

    %% --- Nested Helper Functions ---
    function stepSimulation()
        if ~isvalid(fig) || state.paused
            return;
        end

        for step = 1:state.substeps
            [a1x, a1y, a2x, a2y] = computeAccel(state.x1, state.y1, ...
                                                state.x2, state.y2);

            vx1_half = state.vx1 + 0.5 * a1x * state.dt;
            vy1_half = state.vy1 + 0.5 * a1y * state.dt;
            vx2_half = state.vx2 + 0.5 * a2x * state.dt;
            vy2_half = state.vy2 + 0.5 * a2y * state.dt;

            state.x1 = state.x1 + vx1_half * state.dt;
            state.y1 = state.y1 + vy1_half * state.dt;
            state.x2 = state.x2 + vx2_half * state.dt;
            state.y2 = state.y2 + vy2_half * state.dt;

            [a1x_new, a1y_new, a2x_new, a2y_new] = computeAccel( ...
                state.x1, state.y1, state.x2, state.y2);

            state.vx1 = vx1_half + 0.5 * a1x_new * state.dt;
            state.vy1 = vy1_half + 0.5 * a1y_new * state.dt;
            state.vx2 = vx2_half + 0.5 * a2x_new * state.dt;
            state.vy2 = vy2_half + 0.5 * a2y_new * state.dt;
        end

        updateGraphicsAndTelemetry();
        drawnow limitrate;
    end

    function cleanupAndClose()
        if isvalid(simTimer)
            stop(simTimer);
            delete(simTimer);
        end
        delete(fig);
    end

    function onSliderChange()
        lblMass.Text = sprintf('Mass Ratio (m2 / m1): %.2f', sldMass.Value);
        lblVel.Text  = sprintf('Initial Speed (v / v_circ): %.2f', sldVel.Value);
        resetSimulation();
    end

    function togglePause()
        state.paused = ~state.paused;
        if state.paused
            btnPause.Text = 'Resume';
            btnPause.BackgroundColor = [0.20, 0.65, 0.40];
        else
            btnPause.Text = 'Pause';
            btnPause.BackgroundColor = [0.25, 0.45, 0.75];
        end
    end

    function resetSimulation()
        ratio = sldMass.Value;
        v_mult = sldVel.Value;

        state.m2 = ratio * state.m1;
        M = state.m1 + state.m2;

        state.x1 = -(state.m2 / M) * state.r0;
        state.y1 = 0;
        state.x2 =  (state.m1 / M) * state.r0;
        state.y2 = 0;

        v_circ = sqrt(state.G * M / state.r0);
        v_rel  = v_mult * v_circ;

        state.vx1 = 0;
        state.vy1 = -(state.m2 / M) * v_rel;
        state.vx2 = 0;
        state.vy2 =  (state.m1 / M) * v_rel;

        set(body2, 'MarkerSize', 6 + 8 * ratio);

        clearpoints(trail1);
        clearpoints(trail2);
        updateGraphicsAndTelemetry();
    end

    function [a1x, a1y, a2x, a2y] = computeAccel(x1, y1, x2, y2)
        dx = x2 - x1;
        dy = y2 - y1;
        r2 = dx^2 + dy^2 + 1e-6;
        inv_r3 = r2^(-1.5);

        a1x =  state.G * state.m2 * dx * inv_r3;
        a1y =  state.G * state.m2 * dy * inv_r3;
        a2x = -state.G * state.m1 * dx * inv_r3;
        a2y = -state.G * state.m1 * dy * inv_r3;
    end

    function updateGraphicsAndTelemetry()
        set(body1, 'XData', state.x1, 'YData', state.y1);
        set(body2, 'XData', state.x2, 'YData', state.y2);
        addpoints(trail1, state.x1, state.y1);
        addpoints(trail2, state.x2, state.y2);

        v1_sq = state.vx1^2 + state.vy1^2;
        v2_sq = state.vx2^2 + state.vy2^2;
        K = 0.5 * state.m1 * v1_sq + 0.5 * state.m2 * v2_sq;

        r = hypot(state.x2 - state.x1, state.y2 - state.y1);
        U = -state.G * state.m1 * state.m2 / r;
        E = K + U;

        L = state.m1 * (state.x1 * state.vy1 - state.y1 * state.vx1) + ...
            state.m2 * (state.x2 * state.vy2 - state.y2 * state.vx2);

        if abs(sldVel.Value - 1.0) < 0.02
            regime = 'Circular (E < 0)';
            lblRegime.FontColor = [0.4, 1.0, 0.5];
        elseif E < -1e-4
            regime = 'Bound Elliptical (E < 0)';
            lblRegime.FontColor = [0.4, 0.85, 1.0];
        elseif abs(E) <= 1e-4
            regime = 'Parabolic Escape (E = 0)';
            lblRegime.FontColor = [1.0, 0.8, 0.3];
        else
            regime = 'Hyperbolic Escape (E > 0)';
            lblRegime.FontColor = [1.0, 0.45, 0.45];
        end

        lblRegime.Text = sprintf('Orbit Type: %s', regime);
        lblKE.Text     = sprintf('Kinetic Energy (K):   %+.4f', K);
        lblPE.Text     = sprintf('Potential Energy (U): %+.4f', U);
        lblTE.Text     = sprintf('Total Energy (E):     %+.5f', E);
        lblAngMom.Text = sprintf('Angular Momentum (L): %+.4f', L);
    end
end
